\chapter{Acceptance Tests (UAT)}

Each test consists of at least 7 steps, spans at least two screens, and forms a logical, self-contained scenario. Account creation and login are not included as valid UAT scenarios.

\section{Positive Tests}

\subsection*{UAT-01: Create a New Job}
\begin{table}[H]
  \centering
  \begin{tabularx}{\textwidth}{lX}
    \toprule

    Postconditions & The new job is saved to the database and appears in the job list. \\
    \midrule
  \end{tabularx}
  \begin{tabularx}{\textwidth}{lX}
    \textbf{Steps:} & \\

    4. & Fills in the title, description, and selects a deadline. \\
    5. & Taps on the map to set the job location. \\
    6. & Taps the \textbf{Save} button. \\
    7. & App navigates back to \textbf{Job List}, where the new job appears at the top. \\
    \midrule
    Result & \textbf{PASS \quad/\quad FAIL} \\
    \bottomrule
  \end{tabularx}

\end{table}

\subsection*{UAT-07: Offline Job Status Update}
\begin{table}[H]
  \centering
  \begin{tabularx}{\textwidth}{lX}
    \toprule
    Preconditions  & Worker is logged in with an assigned job. Device has no internet (airplane mode on). \\
    Postconditions & After internet is restored, the status change is synced to the server. \\
    \midrule
  \end{tabularx}
  \begin{tabularx}{\textwidth}{lX}
    \textbf{Steps:} & \\
    1. & Worker enables airplane mode on the device. \\
    2. & Opens \textbf{Job List} --- jobs are still visible (local cache). \\
    3. & Opens \textbf{Job Detail} and taps \textbf{Mark as Done}. \\
    4. & App shows status \textit{Done} with an offline sync icon (clock). \\
    5. & Worker disables airplane mode. \\
    6. & App detects connectivity and triggers background sync. \\
    7. & The clock icon disappears; the job status in \textbf{Job List} is updated on the server. \\
    \midrule
    Result & \textbf{PASS \quad/\quad FAIL} \\
    \bottomrule
  \end{tabularx}
  \caption{UAT-07: Offline job status update}
\end{table}

\section{Negative Tests}

\subsection*{UAT-08: Worker Attempts to Create a Job}
\begin{table}[H]
  \centering
  \begin{tabularx}{\textwidth}{lX}
    \toprule
    Preconditions  & A user with the Worker role is logged in. \\
    Postconditions & App does not allow job creation; server returns a 403 error. \\
    \midrule
  \end{tabularx}
  \begin{tabularx}{\textwidth}{lX}
    \textbf{Steps:} & \\
    1. & Worker opens \textbf{Job List}. \\
    2. & The \textbf{+} (Create Job) button is not visible (hidden at the UI level). \\
    3. & An attacker attempts to call \texttt{POST /api/jobs} directly using the Worker's JWT. \\
    4. & The server verifies the user's role. \\
    5. & Server returns \texttt{403 Forbidden}. \\
    6. & App displays an error message: \textit{``You do not have permission for this action.''}. \\
    7. & No new job is created in the database. \\
    \midrule
    Result & \textbf{PASS \quad/\quad FAIL} \\
    \bottomrule
  \end{tabularx}
  \caption{UAT-08: Unauthorised attempt to create a job (Worker)}
\end{table}
