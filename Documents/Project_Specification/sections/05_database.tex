\section{Entity Descriptions}

\subsection{users}
\begin{table}[H]
  \centering
  \begin{tabularx}{\textwidth}{llX}
    \toprule
    \textbf{Column} & \textbf{Type} & \textbf{Description} \\
    \midrule
    id            & UUID (PK)     & Primary key \\
    email         & VARCHAR(255)  & Unique user email \\
    password\_hash & VARCHAR(255) & Bcrypt hash of password (null for SSO users) \\
    name          & VARCHAR(100)  & Display name \\
    role          & ENUM          & \texttt{manager} or \texttt{worker} \\
    fcm\_token    & TEXT          & Firebase token for push notifications \\
    avatar\_url   & TEXT          & URL of the profile picture \\
    created\_at   & TIMESTAMP     & Record creation time \\
    \bottomrule
  \end{tabularx}
  \caption{Table \texttt{users}}
\end{table}

\subsection{jobs}
\begin{table}[H]
  \centering
  \begin{tabularx}{\textwidth}{llX}
    \toprule
    \textbf{Column} & \textbf{Type} & \textbf{Description} \\
    \midrule
    id            & UUID (PK)     & Primary key \\
    title         & VARCHAR(200)  & Job title \\
    description   & TEXT          & Job description \\
    status        & ENUM          & \texttt{pending}, \texttt{in\_progress}, \texttt{done} \\
    latitude      & FLOAT         & GPS latitude of the job site \\
    longitude     & FLOAT         & GPS longitude of the job site \\
    deadline      & TIMESTAMP     & Completion deadline \\
    created\_by   & UUID (FK)     & Reference to \texttt{users.id} (manager) \\
    created\_at   & TIMESTAMP     & Record creation time \\
    updated\_at   & TIMESTAMP     & Last update time \\
    \bottomrule
  \end{tabularx}
  \caption{Table \texttt{jobs}}
\end{table}
